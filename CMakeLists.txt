cmake_minimum_required(VERSION 3.25)
project(rdr2_the_jockey_club VERSION 0.32.1 LANGUAGES CXX RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(SCRIPHOOK_SDK_ROOT "$ENV{SCRIPHOOK_SDK_ROOT}" CACHE PATH "ScriptHookRDR2 SDK directory")
if(NOT EXISTS "${SCRIPHOOK_SDK_ROOT}/inc/types.h")
    message(FATAL_ERROR "Set SCRIPHOOK_SDK_ROOT to the ScriptHookRDR2 SDK root.")
endif()

add_executable(tjc_installer WIN32 tools/release_installer.cpp tools/clubhouse/portable.manifest)
target_compile_definitions(tjc_installer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(tjc_installer PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(tjc_installer PRIVATE Bcrypt Ole32 Shell32 User32 Uuid)
set_target_properties(tjc_installer PROPERTIES
    OUTPUT_NAME "Install The Jockey Club"
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(race_analysis_export tools/race_analysis/export_course.cpp)
target_include_directories(race_analysis_export PRIVATE "${SCRIPHOOK_SDK_ROOT}/inc")
target_compile_definitions(race_analysis_export PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(race_analysis_export PRIVATE /W4 /EHsc /utf-8 /wd4201 /wd4324)
add_executable(race_settings_catalogue tools/race_analysis/settings_catalogue.cpp)
target_include_directories(race_settings_catalogue PRIVATE "${SCRIPHOOK_SDK_ROOT}/inc")
target_compile_definitions(race_settings_catalogue PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(race_settings_catalogue PRIVATE /W4 /EHsc /utf-8 /wd4201 /wd4324)
add_executable(race_analysis_candidates tools/race_analysis/export_course.cpp)
target_include_directories(race_analysis_candidates PRIVATE "${SCRIPHOOK_SDK_ROOT}/inc")
target_compile_definitions(race_analysis_candidates PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX TJC_ANALYSIS_CANDIDATES)
target_compile_options(race_analysis_candidates PRIVATE /W4 /EHsc /utf-8 /wd4201 /wd4324)
# Portable Clubhouse helpers carry their C++ runtime and use UTF-8 for Windows
# narrow-path APIs. This does not alter the ASI's runtime or game integration.
add_executable(JockeyClub WIN32 tools/clubhouse/launcher.cpp)
target_include_directories(JockeyClub PRIVATE tools/clubhouse)
target_compile_definitions(JockeyClub PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(JockeyClub PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(JockeyClub PRIVATE Winhttp Bcrypt Shell32 Ole32 User32)
add_executable(Clubhouse WIN32 tools/clubhouse/entry.cpp)
target_include_directories(Clubhouse PRIVATE tools/clubhouse)
target_compile_definitions(Clubhouse PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(Clubhouse PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(Clubhouse PRIVATE Shell32 User32)
foreach(clubhouse_target IN ITEMS Clubhouse JockeyClub race_analysis_export race_analysis_candidates race_settings_catalogue)
    set_property(TARGET ${clubhouse_target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_sources(${clubhouse_target} PRIVATE tools/clubhouse/portable.manifest)
endforeach()
add_executable(horse_ledger_seed tools/horse_ledger_seed.cpp src/horse_race_ledger.cpp src/horse_race_ledger_store.cpp)
target_compile_definitions(horse_ledger_seed PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(horse_ledger_seed PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(horse_ledger_seed PRIVATE Bcrypt)
add_executable(horse_ledger_audit tools/horse_ledger_audit.cpp src/horse_race_ledger.cpp src/horse_race_ledger_store.cpp)

target_compile_options(horse_ledger_audit PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(horse_ledger_audit PRIVATE Bcrypt)
add_executable(horse_archive_compact tools/horse_archive_compact.cpp src/horse_race_ledger.cpp src/horse_race_ledger_store.cpp)
target_compile_definitions(horse_archive_compact PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

target_link_libraries(horse_archive_compact PRIVATE Bcrypt)
set_property(TARGET horse_archive_compact PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")


include(CTest)
if(BUILD_TESTING)
    add_library(race_test_natives SHARED tests/native_mock.cpp)
    target_compile_definitions(race_test_natives PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    foreach(case IN ITEMS native_encounter_conditions player_stimulant race_bookmaking race_bookmaker_odds race_bookmaker_camera race_progression race_progression_toast horse_market_catalog race_bookmaker_start race_circuit race_racing_line_joint)
        add_executable(${case}_tests tests/${case}_tests.cpp)
        target_include_directories(${case}_tests PRIVATE "${SCRIPHOOK_SDK_ROOT}/inc" "${CMAKE_SOURCE_DIR}/src")
        target_compile_definitions(${case}_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${case}_tests PRIVATE /EHsc /utf-8 /UNDEBUG /wd4201 /wd4324)
        if(case MATCHES "^(race_progression|race_progression_toast|race_bookmaker_start)$")
            target_link_libraries(${case}_tests PRIVATE race_test_natives)
        endif()
        add_test(NAME ${case} COMMAND ${case}_tests)
        set_tests_properties(${case} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    endforeach()
endif()
